cmake_minimum_required(VERSION 3.25)
project(MaterialCompanion VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(COMPANION_DESKTOP_TESTS "Build focused native workspace tests when available" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Sql)
add_library(companion-core native/core.cpp native/protocol.cpp)
target_include_directories(companion-core PUBLIC native)
target_link_libraries(companion-core PUBLIC Qt6::Core Qt6::Network Qt6::Sql)
add_executable(companion-host native/host.cpp)
target_link_libraries(companion-host PRIVATE companion-core)
add_executable(companion-service native/service.cpp)
target_link_libraries(companion-service PRIVATE companion-core)
if(WIN32)
 target_link_libraries(companion-core PUBLIC advapi32)
endif()
enable_testing()
add_executable(companion-core-tests tests/native/core_tests.cpp native/core.cpp native/protocol.cpp)
target_include_directories(companion-core-tests PRIVATE native)
target_compile_definitions(companion-core-tests PRIVATE COMPANION_TESTING)
target_link_libraries(companion-core-tests PRIVATE Qt6::Core Qt6::Network Qt6::Sql)
if(WIN32)
 target_link_libraries(companion-core-tests PRIVATE advapi32)
endif()
add_test(NAME native-contract COMMAND companion-core-tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/desktop/CMakeLists.txt")
 add_subdirectory(desktop)
else()
 message(STATUS "Desktop shell absent: only native core, service and host are built")
endif()
